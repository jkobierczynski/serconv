# Each test is a small program; test data are generated at run time.
set(SERCONV_TEST_SUPPORT testdata.cpp)

function(serconv_test name)
  add_executable(${name} ${name}.cpp ${SERCONV_TEST_SUPPORT})
  target_link_libraries(${name} PRIVATE serconv_core)
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    target_compile_options(${name} PRIVATE /utf-8)
  endif()
  add_test(NAME ${name} COMMAND ${name} "$<TARGET_FILE:serconv>")
  set_tests_properties(${name} PROPERTIES TIMEOUT 300)
endfunction()

serconv_test(test_time)
serconv_test(test_ser)
serconv_test(test_lj92)
serconv_test(test_mlv)
serconv_test(test_bmff)
serconv_test(test_process)
serconv_test(test_libraw)
serconv_test(test_cli)
if(SERCONV_HAVE_FFMPEG)
  serconv_test(test_ffmpeg)
endif()
